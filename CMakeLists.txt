cmake_minimum_required(VERSION 3.28)
project(GOWA_pc LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(lang C CXX)
    string(REPLACE "/Ob1" "/Ob2" CMAKE_${lang}_FLAGS_RELWITHDEBINFO "${CMAKE_${lang}_FLAGS_RELWITHDEBINFO}")
endforeach()

set(GOWA_GENERATED_DIR "${CMAKE_SOURCE_DIR}/out/ppu" CACHE PATH "Directory with generated PPU sources")
set(GOWA_GENERATED_OPT "-O2" CACHE STRING "Optimization level for generated code")
set(GOWA_SPU_OPT "-O2" CACHE STRING "Optimization level for generated SPU code")

if(MSVC)
    set(GOWA_CLANG /clang:)
    set(GOWA_EH /EHsc)
    set(GOWA_NOEH /GR- /EHs-c-)
else()
    set(GOWA_CLANG "")
    set(GOWA_EH "")
    set(GOWA_NOEH -fno-rtti -fno-exceptions)
endif()
set(GOWA_ARCH_FLAGS ${GOWA_CLANG}-march=x86-64-v3 ${GOWA_CLANG}-ffp-contract=off ${GOWA_CLANG}-fno-strict-aliasing)

add_library(gowa_ppu_ops STATIC runtime/src/ppu_ops.cpp runtime/src/reservation.cpp runtime/src/memory.cpp runtime/src/log.cpp runtime/src/host.cpp)
target_include_directories(gowa_ppu_ops PUBLIC runtime/include "${CMAKE_SOURCE_DIR}/out/include")
target_compile_options(gowa_ppu_ops PRIVATE ${GOWA_ARCH_FLAGS} ${GOWA_EH})

file(GLOB GOWA_RUNTIME_SOURCES CONFIGURE_DEPENDS runtime/src/*.cpp runtime/src/lv2/*.cpp runtime/src/hle/*.cpp runtime/src/media/*.cpp runtime/src/spu/*.cpp runtime/src/spu/edge/*.cpp runtime/src/spu/renjob/*.cpp runtime/src/spu/scream/*.cpp runtime/src/spu/cloth/*.cpp runtime/src/spu/shadows/*.cpp runtime/src/spu/meshprep/*.cpp runtime/src/spu/pm/*.cpp runtime/src/spu/edgejob/*.cpp runtime/src/spu/meshhier/*.cpp runtime/src/spu/anim/*.cpp runtime/src/spu/tasks/*.cpp runtime/src/spu/mpjob/*.cpp runtime/src/jobsys/*.cpp runtime/src/rsx/*.cpp runtime/src/gpu/*.cpp)
list(REMOVE_ITEM GOWA_RUNTIME_SOURCES "${CMAKE_SOURCE_DIR}/runtime/src/ppu_ops.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/main.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/reservation.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/memory.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/log.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/host.cpp")

set(GOWA_SPU_NATIVE_GEN_DIR "${CMAKE_SOURCE_DIR}/out/gen/spu" CACHE PATH "Directory with generated native SPU sources (tools/setup.py)")
set(GOWA_SPU_ROUNDING_SUBDIRS edge renjob scream cloth shadows meshprep jwhole)
if(NOT EXISTS "${GOWA_SPU_NATIVE_GEN_DIR}/generated.txt")
    message(FATAL_ERROR "Generated SPU natives not found in ${GOWA_SPU_NATIVE_GEN_DIR}. They are produced from your own game files: run\n  python tools/setup.py --elf <decrypted patched GOWA.elf> --fw <decrypted 4.91 firmware dir>")
endif()
file(GLOB GOWA_SPU_GEN_SUBDIRS LIST_DIRECTORIES true RELATIVE "${GOWA_SPU_NATIVE_GEN_DIR}" "${GOWA_SPU_NATIVE_GEN_DIR}/*")
set(GOWA_SPU_GEN_SOURCES)
foreach(sub ${GOWA_SPU_GEN_SUBDIRS})
    if(NOT IS_DIRECTORY "${GOWA_SPU_NATIVE_GEN_DIR}/${sub}")
        continue()
    endif()
    file(GLOB gen_srcs CONFIGURE_DEPENDS "${GOWA_SPU_NATIVE_GEN_DIR}/${sub}/*.cpp")
    file(GLOB own_srcs CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/${sub}/*.cpp")
    set_source_files_properties(${gen_srcs} PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/runtime/src/spu/${sub}")
    set_source_files_properties(${own_srcs} PROPERTIES INCLUDE_DIRECTORIES "${GOWA_SPU_NATIVE_GEN_DIR}/${sub};${CMAKE_SOURCE_DIR}/runtime/src/spu/${sub}")
    if(sub IN_LIST GOWA_SPU_ROUNDING_SUBDIRS)
        set_source_files_properties(${gen_srcs} PROPERTIES COMPILE_OPTIONS "${GOWA_CLANG}-frounding-math")
    endif()
    list(APPEND GOWA_SPU_GEN_SOURCES ${gen_srcs})
endforeach()
file(GLOB GOWA_JWHOLE_GEN CONFIGURE_DEPENDS "${GOWA_SPU_NATIVE_GEN_DIR}/jwhole/*.cpp")
if(GOWA_JWHOLE_GEN)
    set_source_files_properties(${GOWA_JWHOLE_GEN} PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/runtime/src/spu/renjob;${CMAKE_SOURCE_DIR}/runtime/src/jobsys")
endif()
file(GLOB GOWA_JOBSYS_OWN CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/jobsys/*.cpp")
set_source_files_properties(${GOWA_JOBSYS_OWN} PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/runtime/src;${CMAKE_SOURCE_DIR}/runtime/src/jobsys;${CMAKE_SOURCE_DIR}/runtime/src/spu/renjob;${GOWA_SPU_NATIVE_GEN_DIR}/jwhole")
list(APPEND GOWA_RUNTIME_SOURCES ${GOWA_SPU_GEN_SOURCES})
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glslang EXCLUDE_FROM_ALL)

add_library(gowa_volk OBJECT third_party/volk/volk.c)
target_include_directories(gowa_volk PUBLIC third_party/volk third_party/Vulkan-Headers/include)
if(WIN32)
    target_compile_definitions(gowa_volk PUBLIC VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES)
    set(GOWA_SYS_LIBS dbghelp winmm)
else()
    target_compile_definitions(gowa_volk PUBLIC VK_NO_PROTOTYPES)
    find_package(SDL3 REQUIRED CONFIG)
    find_package(PkgConfig)
    if(PKG_CONFIG_FOUND)
        pkg_check_modules(GOWA_AVCODEC IMPORTED_TARGET libavcodec libavutil)
    endif()
    set(GOWA_SYS_LIBS SDL3::SDL3 ${CMAKE_DL_LIBS} pthread)
    if(GOWA_AVCODEC_FOUND)
        message(STATUS "libavcodec found, cellVdec HLE enabled")
        list(APPEND GOWA_SYS_LIBS PkgConfig::GOWA_AVCODEC)
    else()
        message(STATUS "libavcodec not found, cellVdec falls back to LLE")
    endif()
endif()

add_library(gowa_runtime OBJECT ${GOWA_RUNTIME_SOURCES})
file(GLOB GOWA_EDGE_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/edge/*.cpp")
file(GLOB GOWA_RENJOB_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/renjob/*.cpp")
file(GLOB GOWA_SCREAM_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/scream/*.cpp")
file(GLOB GOWA_CLOTH_SHADOWS_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/cloth/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/shadows/*.cpp")
file(GLOB GOWA_MESHPREP_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/meshprep/*.cpp")
file(GLOB GOWA_PM_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/pm/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/edgejob/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/meshhier/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/anim/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/tasks/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/mpjob/*.cpp")
set_source_files_properties("${CMAKE_SOURCE_DIR}/runtime/src/spu/texcreator_native.cpp" ${GOWA_EDGE_SOURCES} ${GOWA_RENJOB_SOURCES} ${GOWA_SCREAM_SOURCES} ${GOWA_CLOTH_SHADOWS_SOURCES} ${GOWA_MESHPREP_SOURCES} PROPERTIES COMPILE_OPTIONS "${GOWA_CLANG}-frounding-math")
target_include_directories(gowa_runtime PRIVATE third_party/volk third_party/Vulkan-Headers/include third_party/glslang "${CMAKE_BINARY_DIR}/include")
target_compile_definitions(gowa_runtime PRIVATE VK_NO_PROTOTYPES)
target_link_libraries(gowa_runtime PUBLIC gowa_ppu_ops)
target_compile_options(gowa_runtime PRIVATE ${GOWA_ARCH_FLAGS} ${GOWA_EH})
if(NOT WIN32)
    target_link_libraries(gowa_runtime PRIVATE SDL3::SDL3)
    if(GOWA_AVCODEC_FOUND)
        target_link_libraries(gowa_runtime PRIVATE PkgConfig::GOWA_AVCODEC)
        target_compile_definitions(gowa_runtime PRIVATE GOWA_HAVE_FFMPEG)
    endif()
endif()

file(GLOB GOWA_GENERATED_SOURCES CONFIGURE_DEPENDS "${GOWA_GENERATED_DIR}/*_[0-9][0-9][0-9].cpp")
add_library(gowa_ppu STATIC ${GOWA_GENERATED_SOURCES} "${GOWA_GENERATED_DIR}/table.cpp")
target_include_directories(gowa_ppu PRIVATE "${GOWA_GENERATED_DIR}")
target_link_libraries(gowa_ppu PUBLIC gowa_ppu_ops)
target_compile_options(gowa_ppu PRIVATE ${GOWA_ARCH_FLAGS} ${GOWA_CLANG}${GOWA_GENERATED_OPT} ${GOWA_NOEH} ${GOWA_CLANG}-Wno-unused-label ${GOWA_CLANG}-Wno-unused-variable)
set_source_files_properties(${GOWA_GENERATED_SOURCES} PROPERTIES SKIP_PRECOMPILE_HEADERS ON)

set(GOWA_SPU_GENERATED_DIR "${CMAKE_SOURCE_DIR}/out/spu" CACHE PATH "Directory with generated SPU sources")
if(NOT EXISTS "${GOWA_SPU_GENERATED_DIR}/spu_registry.cpp")
    message(FATAL_ERROR "SPU AOT sources not found in ${GOWA_SPU_GENERATED_DIR}. Run: python tools/setup.py")

target_include_directories(gowa_spu PRIVATE runtime/include runtime/src/spu)

target_compile_options(gowa_spu PRIVATE ${GOWA_ARCH_FLAGS} ${GOWA_CLANG}${GOWA_SPU_OPT} ${GOWA_CLANG}-frounding-math ${GOWA_NOEH})

set(GOWA_STREAMLINE_DIR "${CMAKE_SOURCE_DIR}/third_party/streamline" CACHE PATH "Extracted NVIDIA Streamline SDK (optional, enables GOWA_DLSSG)")
if(WIN32 AND EXISTS "${GOWA_STREAMLINE_DIR}/include/sl.h" AND EXISTS "${GOWA_STREAMLINE_DIR}/lib/x64/sl.interposer.lib")
    message(STATUS "Streamline SDK: ${GOWA_STREAMLINE_DIR}")
    target_compile_definitions(gowa_runtime PRIVATE GOWA_HAVE_STREAMLINE)
    if(NOT EXISTS "${GOWA_STREAMLINE_DIR}/external/ngx-sdk/include/nvsdk_ngx_params.h")
        target_compile_definitions(gowa_runtime PRIVATE GOWA_DLSS_DEFAULT_DIR="${CMAKE_SOURCE_DIR}/dlss")
    endif()
    target_include_directories(gowa_runtime PRIVATE "${GOWA_STREAMLINE_DIR}/include")
    foreach(t gowa spu_live)
        target_link_libraries(${t} PRIVATE "${GOWA_STREAMLINE_DIR}/lib/x64/sl.interposer.lib" delayimp)
        target_link_options(${t} PRIVATE /DELAYLOAD:sl.interposer.dll)
    endforeach()
    set(GOWA_SL_DLLS sl.interposer.dll sl.common.dll sl.dlss_g.dll sl.reflex.dll sl.pcl.dll NvLowLatencyVk.dll)
    foreach(dll ${GOWA_SL_DLLS})
        add_custom_command(TARGET gowa POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GOWA_STREAMLINE_DIR}/bin/x64/${dll}" "$<TARGET_FILE_DIR:gowa>/${dll}")
    endforeach()
else()
    message(STATUS "Streamline SDK not found, DLSS frame generation disabled")
endif()

set(GOWA_NGX_INCLUDE "${GOWA_STREAMLINE_DIR}/external/ngx-sdk/include")
if(WIN32 AND EXISTS "${GOWA_NGX_INCLUDE}/nvsdk_ngx_params.h" AND EXISTS "${GOWA_NGX_INCLUDE}/nvsdk_ngx_defs_vk.h")
    message(STATUS "NGX SDK headers: ${GOWA_NGX_INCLUDE}")
    target_compile_definitions(gowa_runtime PRIVATE GOWA_HAVE_NGX GOWA_DLSS_DEFAULT_DIR="${CMAKE_SOURCE_DIR}/dlss")
    target_include_directories(gowa_runtime PRIVATE "${GOWA_NGX_INCLUDE}")
    add_library(gowa_nvngx SHARED runtime/ngx_bridge/bridge.c)
    set_target_properties(gowa_nvngx PROPERTIES OUTPUT_NAME gowa_nvngx)
    foreach(t gowa spu_live)
        add_dependencies(${t} gowa_nvngx)
        target_link_libraries(${t} PRIVATE version)
    endforeach()
else()
    message(STATUS "NGX SDK headers not found, DLSS neural rendering disabled")
endif()
