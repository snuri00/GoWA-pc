cmake_minimum_required(VERSION 3.28)
project(GOWA_pc LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(lang C CXX)
    string(REPLACE "/Ob1" "/Ob2" CMAKE_${lang}_FLAGS_RELWITHDEBINFO "${CMAKE_${lang}_FLAGS_RELWITHDEBINFO}")
endforeach()

set(GOWA_GENERATED_DIR "${CMAKE_SOURCE_DIR}/out/ppu" CACHE PATH "Directory with generated PPU sources")
set(GOWA_GENERATED_OPT "-O2" CACHE STRING "Optimization level for generated code")
set(GOWA_SPU_OPT "-O2" CACHE STRING "Optimization level for generated SPU code")

set(GOWA_ARCH_FLAGS /clang:-march=x86-64-v3 /clang:-ffp-contract=off /clang:-fno-strict-aliasing)

add_library(gowa_ppu_ops STATIC runtime/src/ppu_ops.cpp runtime/src/reservation.cpp runtime/src/memory.cpp runtime/src/log.cpp)
target_include_directories(gowa_ppu_ops PUBLIC runtime/include "${CMAKE_SOURCE_DIR}/out/include")
target_compile_options(gowa_ppu_ops PRIVATE ${GOWA_ARCH_FLAGS} /EHsc)

file(GLOB GOWA_RUNTIME_SOURCES CONFIGURE_DEPENDS runtime/src/*.cpp runtime/src/lv2/*.cpp runtime/src/hle/*.cpp runtime/src/spu/*.cpp runtime/src/spu/edge/*.cpp runtime/src/spu/renjob/*.cpp runtime/src/spu/scream/*.cpp runtime/src/spu/cloth/*.cpp runtime/src/spu/shadows/*.cpp runtime/src/spu/meshprep/*.cpp runtime/src/spu/pm/*.cpp runtime/src/rsx/*.cpp runtime/src/gpu/*.cpp)
list(REMOVE_ITEM GOWA_RUNTIME_SOURCES "${CMAKE_SOURCE_DIR}/runtime/src/ppu_ops.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/main.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/reservation.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/memory.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/log.cpp")
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glslang EXCLUDE_FROM_ALL)

add_library(gowa_volk OBJECT third_party/volk/volk.c)
target_include_directories(gowa_volk PUBLIC third_party/volk third_party/Vulkan-Headers/include)
target_compile_definitions(gowa_volk PUBLIC VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES)

add_library(gowa_runtime OBJECT ${GOWA_RUNTIME_SOURCES})
file(GLOB GOWA_EDGE_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/edge/*.cpp")
file(GLOB GOWA_RENJOB_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/renjob/*.cpp")
file(GLOB GOWA_SCREAM_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/scream/*.cpp")
file(GLOB GOWA_CLOTH_SHADOWS_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/cloth/*.cpp" "${CMAKE_SOURCE_DIR}/runtime/src/spu/shadows/*.cpp")
file(GLOB GOWA_MESHPREP_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/meshprep/*.cpp")
file(GLOB GOWA_PM_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/src/spu/pm/*.cpp")
set_source_files_properties("${CMAKE_SOURCE_DIR}/runtime/src/spu/texcreator_native.cpp" ${GOWA_EDGE_SOURCES} ${GOWA_RENJOB_SOURCES} ${GOWA_SCREAM_SOURCES} ${GOWA_CLOTH_SHADOWS_SOURCES} ${GOWA_MESHPREP_SOURCES} PROPERTIES COMPILE_OPTIONS "/clang:-frounding-math")
target_include_directories(gowa_runtime PRIVATE third_party/volk third_party/Vulkan-Headers/include third_party/glslang "${CMAKE_BINARY_DIR}/include")
target_compile_definitions(gowa_runtime PRIVATE VK_NO_PROTOTYPES)
target_link_libraries(gowa_runtime PUBLIC gowa_ppu_ops)
target_compile_options(gowa_runtime PRIVATE ${GOWA_ARCH_FLAGS} /EHsc)

file(GLOB GOWA_GENERATED_SOURCES CONFIGURE_DEPENDS "${GOWA_GENERATED_DIR}/*_[0-9][0-9][0-9].cpp")
add_library(gowa_ppu STATIC ${GOWA_GENERATED_SOURCES} "${GOWA_GENERATED_DIR}/table.cpp")
target_include_directories(gowa_ppu PRIVATE "${GOWA_GENERATED_DIR}")
target_link_libraries(gowa_ppu PUBLIC gowa_ppu_ops)
target_compile_options(gowa_ppu PRIVATE ${GOWA_ARCH_FLAGS} /clang:${GOWA_GENERATED_OPT} /GR- /EHs-c- /clang:-Wno-unused-label /clang:-Wno-unused-variable)
set_source_files_properties(${GOWA_GENERATED_SOURCES} PROPERTIES SKIP_PRECOMPILE_HEADERS ON)

set(GOWA_SPU_GENERATED_DIR "${CMAKE_SOURCE_DIR}/out/spu" CACHE PATH "Directory with generated SPU sources")
if(NOT EXISTS "${GOWA_SPU_GENERATED_DIR}/spu_registry.cpp")
    execute_process(COMMAND python "${CMAKE_SOURCE_DIR}/recomp/spu/codegen.py" "${GOWA_SPU_GENERATED_DIR}")
endif()
file(GLOB GOWA_SPU_SOURCES CONFIGURE_DEPENDS "${GOWA_SPU_GENERATED_DIR}/spu_*.cpp")
add_library(gowa_spu STATIC ${GOWA_SPU_SOURCES})
target_include_directories(gowa_spu PRIVATE runtime/include runtime/src/spu)
target_link_libraries(gowa_spu PUBLIC gowa_ppu_ops)
target_compile_options(gowa_spu PRIVATE ${GOWA_ARCH_FLAGS} /clang:${GOWA_SPU_OPT} /clang:-frounding-math /GR- /EHs-c-)

add_executable(gowa runtime/src/main.cpp $<TARGET_OBJECTS:gowa_runtime> $<TARGET_OBJECTS:gowa_volk>)
target_link_libraries(gowa PRIVATE gowa_ppu gowa_spu gowa_ppu_ops dbghelp winmm glslang SPIRV glslang-default-resource-limits)
target_include_directories(gowa PRIVATE runtime/include)
target_compile_options(gowa PRIVATE ${GOWA_ARCH_FLAGS} /EHsc)

if(EXISTS "${CMAKE_SOURCE_DIR}/out/conformance/tests.cpp")
    add_executable(ppu_conformance tests/ppu_conformance/harness.cpp "${CMAKE_SOURCE_DIR}/out/conformance/tests.cpp")
    target_include_directories(ppu_conformance PRIVATE tests/ppu_conformance)
    target_link_libraries(ppu_conformance PRIVATE gowa_ppu_ops)
    target_compile_options(ppu_conformance PRIVATE ${GOWA_ARCH_FLAGS} /clang:${GOWA_GENERATED_OPT})
endif()

add_executable(spu_bench tests/spu_bench/bench.cpp runtime/src/spu/interpreter.cpp runtime/src/spu/aot.cpp runtime/src/spu/hle.cpp runtime/src/spu/hle_builtin.cpp runtime/src/spu/texcreator_native.cpp ${GOWA_EDGE_SOURCES} ${GOWA_RENJOB_SOURCES} ${GOWA_SCREAM_SOURCES} ${GOWA_CLOTH_SHADOWS_SOURCES} ${GOWA_MESHPREP_SOURCES} ${GOWA_PM_SOURCES} runtime/src/log.cpp)
target_include_directories(spu_bench PRIVATE runtime/include runtime/src/spu)
target_link_libraries(spu_bench PRIVATE gowa_spu gowa_ppu_ops dbghelp)
target_compile_options(spu_bench PRIVATE ${GOWA_ARCH_FLAGS} /clang:-frounding-math /EHsc)

add_executable(spu_conformance tests/spu_conformance/harness.cpp runtime/src/spu/interpreter.cpp)
target_include_directories(spu_conformance PRIVATE runtime/include)
target_link_libraries(spu_conformance PRIVATE gowa_ppu_ops)
target_compile_options(spu_conformance PRIVATE ${GOWA_ARCH_FLAGS} /EHsc)
